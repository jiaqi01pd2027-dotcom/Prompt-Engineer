```python
from __future__ import annotations

import re

_ISO_DURATION_RE = re.compile(
    r"^P"
    r"(?:(?P<years>\d+)Y)?"
    r"(?:(?P<months>\d+)M)?"
    r"(?:(?P<weeks>\d+)W)?"
    r"(?:(?P<days>\d+)D)?"
    r"(?:T"
    r"(?:(?P<hours>\d+)H)?"
    r"(?:(?P<minutes>\d+)M)?"
    r"(?:(?P<seconds>\d+(?:\.\d+)?)S)?"
    r")?$"
)

_UNIT_SECONDS = {
    "years": 365 * 86400,
    "months": 30 * 86400,
    "weeks": 7 * 86400,
    "days": 86400,
    "hours": 3600,
    "minutes": 60,
    "seconds": 1,
}


def parse_iso_duration(s: str) -> float | int:
    """Convert an ISO-8601 duration string to a total number of seconds.

    Unit assumptions:
        1 year   = 365 days
        1 month  = 30 days
        1 week   = 7 days
        1 day    = 24 hours
        1 hour   = 60 minutes
        1 minute = 60 seconds

    Returns an int when the string has no fractional part, otherwise a float.
    Raises ValueError for anything that is not a well-formed duration: missing
    leading "P", no components at all ("P", "PT", "P1DT"), time units without
    the "T" separator ("P1H"), fractions on any unit other than seconds
    ("P1.5D"), weeks combined with other units ("P1W2D"), or unknown
    characters (lowercase, signs, whitespace, ",", etc.).
    """
    if not isinstance(s, str):
        raise ValueError(f"duration must be a str, got {type(s).__name__}")

    m = _ISO_DURATION_RE.fullmatch(s)
    if m is None:
        raise ValueError(f"invalid ISO-8601 duration: {s!r}")

    parts = {k: v for k, v in m.groupdict().items() if v is not None}
    if not parts:
        raise ValueError(f"duration has no components: {s!r}")
    if s.endswith("T"):
        raise ValueError(f"'T' separator with no time components: {s!r}")
    if "weeks" in parts and len(parts) > 1:
        raise ValueError(f"weeks cannot be combined with other units: {s!r}")

    total = 0
    for unit, value in parts.items():
        if unit == "seconds" and "." in value:
            total += float(value)
        else:
            total += int(value) * _UNIT_SECONDS[unit]
    return total
```

Assumptions
- Components must appear in ISO order (Y, M, W, D, then T, H, M, S), each at most once; `_ISO_DURATION_RE` and `_UNIT_SECONDS` are module-level helpers, not extra public functions.
- Weeks are rejected when combined with any other unit (`P1W2D`), matching "weeks always appear alone".
- Only `.` is accepted as the decimal separator on seconds; the ISO-permitted `,` form is treated as an unknown character.
- Zero-valued durations such as `PT0S` or `P0D` are valid and return `0`; the return type is `int` unless seconds carry a fraction, in which case it is `float`.
- Non-string inputs raise `ValueError` (not `TypeError`) so a single exception type guards the database path; `from __future__ import annotations` keeps the exact `float | int` signature working on Python 3.7+.
