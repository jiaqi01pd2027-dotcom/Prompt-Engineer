```python
from __future__ import annotations
import re

_DURATION_RE = re.compile(
    r"^P"
    r"(?:(?P<Y>\d+)Y)?"
    r"(?:(?P<Mo>\d+)M)?"
    r"(?:(?P<W>\d+)W)?"
    r"(?:(?P<D>\d+)D)?"
    r"(?:T"
    r"(?:(?P<H>\d+)H)?"
    r"(?:(?P<Mi>\d+)M)?"
    r"(?:(?P<S>\d+(?:[.,]\d+)?)S)?"
    r")?$"
)

_SECONDS_PER = {
    "Y": 365 * 86400,
    "Mo": 30 * 86400,
    "W": 7 * 86400,
    "D": 86400,
    "H": 3600,
    "Mi": 60,
    "S": 1,
}


def parse_iso_duration(s: str) -> float | int:
    """Convert an ISO 8601 duration such as "P1DT12H" into seconds.

    Units are fixed-length: 1 year = 365 days, 1 month = 30 days,
    1 week = 7 days, 1 day = 24 hours. A fractional value is only
    allowed for seconds. Returns an int unless seconds are fractional.
    Raises ValueError for anything that is not a valid duration.
    """
    if not isinstance(s, str):
        raise ValueError(f"expected a string, got {type(s).__name__}")

    m = _DURATION_RE.match(s)
    if m is None:
        raise ValueError(f"invalid ISO 8601 duration: {s!r}")

    parts = {k: v for k, v in m.groupdict().items() if v is not None}
    if not parts:
        # "P" or "PT" with no components
        raise ValueError(f"invalid ISO 8601 duration: {s!r}")
    if "T" in s and not (parts.keys() & {"H", "Mi", "S"}):
        # "P1DT" -- T separator with nothing after it
        raise ValueError(f"invalid ISO 8601 duration: {s!r}")

    total = 0
    for key, value in parts.items():
        if key == "S" and ("." in value or "," in value):
            total += float(value.replace(",", ".")) * _SECONDS_PER[key]
        else:
            total += int(value) * _SECONDS_PER[key]
    return total
```

**How it works**

- A single anchored regex encodes the grammar `P[nY][nM][nW][nD][T[nH][nM][nS]]`. Because the `H`/`M`/`S` groups live inside the optional `T...` block, a time unit without the `T` separator (`P1H`, `P1S`) simply fails to match, and so does a date unit after `T` (`PT1D`). Only the seconds group allows a decimal part, so `P1.5D` or `PT1.5H` are rejected too. Order is enforced (`P1D2Y` fails), and unknown characters, leading/trailing whitespace, lowercase, and negatives are all rejected by the anchors.
- Two extra checks cover cases the regex alone accepts: bare `P`/`PT` (nothing matched) and a trailing `T` with no time components (`P1DT`).
- Result type: `int` unless seconds carry a fraction, in which case `float`. ISO 8601 allows a comma as the decimal mark, so `PT1,5S` is accepted alongside `PT1.5S`.
