```python
"""Parse ISO-8601 duration strings into seconds using only the standard library."""

from __future__ import annotations

import re

_SECONDS_PER = {
    "Y": 365 * 86400,
    "M": 30 * 86400,
    "W": 7 * 86400,
    "D": 86400,
    "H": 3600,
    "TM": 60,  # minutes: the "M" that appears after the T separator
    "S": 1,
}

# Anchored, strict grammar. Every component is optional here; emptiness and the
# "T with nothing after it" case are rejected after the match.
_DURATION_RE = re.compile(
    r"""
    ^P
    (?:(?P<Y>\d+)Y)?
    (?:(?P<M>\d+)M)?
    (?:(?P<W>\d+)W)?
    (?:(?P<D>\d+)D)?
    (?:T
        (?:(?P<H>\d+)H)?
        (?:(?P<TM>\d+)M)?
        (?:(?P<S>\d+(?:[.,]\d+)?)S)?
    )?
    $
    """,
    re.VERBOSE,
)


def parse_iso_duration(s: str) -> float | int:
    """Convert an ISO-8601 duration such as ``"P1Y2M3DT4H5M6S"`` to seconds.

    Unit lengths: 1 year = 365 days, 1 month = 30 days, 1 week = 7 days,
    1 day = 24 hours. Fractions are only accepted in the seconds component
    (``"PT1.5S"``); a decimal comma is accepted as ISO-8601 allows it.

    Returns an ``int`` when every component is integral and a ``float`` when
    the seconds component carries a fraction.

    Raises:
        ValueError: if ``s`` is not a valid ISO-8601 duration.
    """
    if not isinstance(s, str):
        raise ValueError(f"expected str, got {type(s).__name__}")

    match = _DURATION_RE.fullmatch(s)
    if match is None:
        raise ValueError(f"invalid ISO-8601 duration: {s!r}")

    parts = {k: v for k, v in match.groupdict().items() if v is not None}
    if not parts:
        # Matches bare "P" and "PT": the grammar allows them, the standard does not.
        raise ValueError(f"invalid ISO-8601 duration: {s!r} has no components")
    if "T" in s and not (parts.keys() & {"H", "TM", "S"}):
        raise ValueError(f"invalid ISO-8601 duration: {s!r} has 'T' with no time components")

    total = 0
    for key, value in parts.items():
        if key == "S" and ("." in value or "," in value):
            total += float(value.replace(",", ".")) * _SECONDS_PER[key]
        else:
            total += int(value) * _SECONDS_PER[key]
    return total
```

**Design notes**

- A single anchored regex encodes the grammar, so ordering (`Y M W D T H M S`), repeats, and the "which M is minutes" ambiguity are all handled by structure rather than by a hand-written scanner. `fullmatch` rather than `$` is deliberate: `$` would silently accept a trailing newline.
- The two cases the regex can't reject on its own (bare `P`/`PT`, and `P1DT` with a dangling `T`) are checked explicitly after the match.
- Fractions are only grammatically possible in the seconds group, so `P1.5D` and `PT1.5H` fail at the regex. The decimal comma is accepted because ISO-8601 permits it.
- Return type is `int` for integral inputs and `float` only when seconds carry a fraction, matching the `float | int` signature without forcing everything to float.
- Negative durations and whitespace are rejected. Both are outside the spec's grammar, and negatives aren't in ISO-8601 proper (only in the ISO 8601-2 extension).
