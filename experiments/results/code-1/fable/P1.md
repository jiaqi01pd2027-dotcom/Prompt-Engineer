```python
from __future__ import annotations

import re

_ISO_DURATION = re.compile(
    r"""^P
    (?!$)                                   # "P" alone is invalid
    (?:(?P<weeks>\d+)W)?
    (?:(?P<years>\d+)Y)?
    (?:(?P<months>\d+)M)?
    (?:(?P<days>\d+)D)?
    (?:T
        (?=\d)                              # "T" must be followed by at least one component
        (?:(?P<hours>\d+)H)?
        (?:(?P<minutes>\d+)M)?
        (?:(?P<seconds>\d+(?:[.,]\d+)?)S)?
    )?
    $""",
    re.VERBOSE,
)

_SECONDS_PER = {
    "years": 365 * 86400,
    "months": 30 * 86400,
    "weeks": 7 * 86400,
    "days": 86400,
    "hours": 3600,
    "minutes": 60,
    "seconds": 1,
}


def parse_iso_duration(s: str) -> float | int:
    """Convert an ISO-8601 duration string (e.g. "P1Y2M3DT4H5M6S") to seconds.

    Uses 365 days/year, 30 days/month, 7 days/week, 24 hours/day.
    Returns an int unless the seconds component is fractional, in which case a float.
    Raises ValueError for anything that is not a valid ISO-8601 duration.
    """
    if not isinstance(s, str):
        raise ValueError(f"expected str, got {type(s).__name__}")

    m = _ISO_DURATION.match(s)
    if m is None:
        raise ValueError(f"invalid ISO-8601 duration: {s!r}")

    total = 0
    for unit, value in m.groupdict().items():
        if value is None:
            continue
        if unit == "seconds" and ("." in value or "," in value):
            total += float(value.replace(",", ".")) * _SECONDS_PER[unit]
        else:
            total += int(value) * _SECONDS_PER[unit]
    return total
```

**How it works**

A single anchored regex enforces the grammar: leading `P`, optional `W`/`Y`/`M`/`D` in that order, then an optional `T` block with `H`/`M`/`S`. Two lookaheads reject the degenerate cases: `(?!$)` after `P` rejects a bare `"P"`, and `(?=\d)` after `T` rejects a trailing `"T"` with nothing following. Because the date-side `M` and time-side `M` live in different groups, `PT5M` is minutes and `P5M` is months with no ambiguity. Only the seconds group permits a fraction (`.` or `,`, both allowed by the standard), so `P1.5D` and `PT1.5H` fail to match and raise `ValueError`.

**Return type**: an `int` whenever every component is an integer (`"PT36H"` returns `129600`), and a `float` only when seconds are fractional (`"PT1.5S"` returns `1.5`), matching the `float | int` signature.
