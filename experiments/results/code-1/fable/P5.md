```python
from __future__ import annotations

import re

_ISO_DURATION = re.compile(
    r"^P"
    r"(?:(?P<years>\d+)Y)?"
    r"(?:(?P<months>\d+)M)?"
    r"(?:(?P<weeks>\d+)W)?"
    r"(?:(?P<days>\d+)D)?"
    r"(?:T"
    r"(?:(?P<hours>\d+)H)?"
    r"(?:(?P<minutes>\d+)M)?"
    r"(?:(?P<seconds>\d+(?:[.,]\d+)?)S)?"
    r")?$"
)

_SECONDS_PER = {
    "years": 365 * 86400,
    "months": 30 * 86400,
    "weeks": 7 * 86400,
    "days": 86400,
    "hours": 3600,
    "minutes": 60,
    "seconds": 1,
}


def parse_iso_duration(s: str) -> float | int:
    """Convert an ISO-8601 duration string to a total number of seconds.

    Unit assumptions:
        1 year   = 365 days
        1 month  = 30 days
        1 week   = 7 days
        1 day    = 24 hours
        1 hour   = 60 minutes
        1 minute = 60 seconds

    Returns an int when the result is whole, otherwise a float (only a
    fractional seconds component can produce a float).

    Raises:
        ValueError: if ``s`` is not a valid ISO-8601 duration.
    """
    if not isinstance(s, str):
        raise ValueError(f"expected str, got {type(s).__name__}")
    m = _ISO_DURATION.fullmatch(s)
    if m is None:
        raise ValueError(f"invalid ISO-8601 duration: {s!r}")
    parts = {k: v for k, v in m.groupdict().items() if v is not None}
    if not parts:
        raise ValueError(f"ISO-8601 duration has no components: {s!r}")
    if s.endswith("T"):
        raise ValueError(f"'T' separator with no time components: {s!r}")
    total = 0
    for unit, value in parts.items():
        if unit == "seconds" and ("." in value or "," in value):
            total += float(value.replace(",", ".")) * _SECONDS_PER[unit]
        else:
            total += int(value) * _SECONDS_PER[unit]
    return total
```

**Assumptions**
- Designators must be uppercase and appear in canonical order (Y, M, W, D, then T, H, M, S); lowercase or reordered input raises ValueError.
- Weeks may be combined with other date components (e.g. "P1Y2W") rather than being strictly standalone, since the sum is unambiguous.
- Fractions are accepted only in the seconds component, with either "." or "," as the decimal mark; a fraction elsewhere raises ValueError.
- Whole-number results are returned as int; a fractional seconds component yields a float.
- Negative durations, leading/trailing whitespace, and "P"/"PT" with no components are rejected.
