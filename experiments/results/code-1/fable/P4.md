```python
import re

_DURATION_RE = re.compile(
    r"^P(?!$)"                                   # leading P, but not bare "P"
    r"(?:"
    r"(?P<weeks>\d+)W"                           # standalone weeks form: PnW
    r"|"
    r"(?:(?P<years>\d+)Y)?"
    r"(?:(?P<months>\d+)M)?"
    r"(?:(?P<days>\d+)D)?"
    r"(?:T(?=\d)"                                # T must be followed by a component
    r"(?:(?P<hours>\d+)H)?"
    r"(?:(?P<minutes>\d+)M)?"
    r"(?:(?P<seconds>\d+(?:[.,]\d+)?)S)?"        # fraction allowed only on seconds
    r")?"
    r")$"
)

_SECONDS_PER = {
    "years": 365 * 86400,
    "months": 30 * 86400,
    "weeks": 7 * 86400,
    "days": 86400,
    "hours": 3600,
    "minutes": 60,
    "seconds": 1,
}


def parse_iso_duration(s):
    """Convert an ISO-8601 duration string (e.g. "P1Y2M3DT4H5M6S", "P2W",
    "PT1.5S") to a total number of seconds.

    Uses 365 days per year and 30 days per month. Returns an int when the
    duration has no fractional seconds, otherwise a float.

    Raises ValueError for anything that is not a valid ISO-8601 duration.
    """
    if not isinstance(s, str):
        raise ValueError(f"expected an ISO-8601 duration string, got {type(s).__name__}")

    match = _DURATION_RE.fullmatch(s)
    if match is None:
        raise ValueError(f"invalid ISO-8601 duration: {s!r}")

    parts = {k: v for k, v in match.groupdict().items() if v is not None}
    if not parts:
        # Matched the regex but carried no components (e.g. "PT").
        raise ValueError(f"invalid ISO-8601 duration: {s!r}")

    total = 0
    for name, value in parts.items():
        if name == "seconds" and ("." in value or "," in value):
            total += float(value.replace(",", "."))
        else:
            total += int(value) * _SECONDS_PER[name]
    return total
```

**How it works**

- A single anchored regex does the validation. The `(?!$)` after `P` rejects bare `"P"`; `T(?=\d)` requires the time part to actually contain something, so `"PT"` and `"P1DT"` fail. Since H/M/S groups only live inside the `T...` branch, `"P1H"` or `"P1S"` cannot match.
- Only the seconds group permits a fraction (`.` or `,`, both valid in ISO-8601), so `"P1.5D"` and `"PT1.5H"` raise.
- Component order is enforced (`"P1D1Y"` raises), and the regex is anchored with `fullmatch`, so leading/trailing whitespace, lowercase, or stray characters raise.
- Returns an `int` for whole-second durations (e.g. `"PT36H"` -> `129600`) and a `float` only when fractional seconds are present (`"PT1.5S"` -> `1.5`).
