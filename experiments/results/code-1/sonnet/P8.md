```python
import re

_WEEK_RE = re.compile(r"P(\d+)W")
_DURATION_RE = re.compile(
    r"P"
    r"(?:(?P<years>\d+)Y)?"
    r"(?:(?P<months>\d+)M)?"
    r"(?:(?P<days>\d+)D)?"
    r"(?:T"
    r"(?:(?P<hours>\d+)H)?"
    r"(?:(?P<minutes>\d+)M)?"
    r"(?:(?P<seconds>\d+(?:\.\d+)?)S)?"
    r")?"
)


def parse_iso_duration(s: str) -> float | int:
    """Convert an ISO-8601 duration string to a total number of seconds.

    Unit assumptions (fixed by the billing service, not calendar-accurate):
      - 1 year  = 365 days
      - 1 month = 30 days
      - 1 week  = 7 days  (weeks only ever appear alone, as "P<n>W")
      - 1 day   = 24 hours
      - Fractional values are only accepted on the seconds component
        (e.g. "PT1.5S"); every other component must be an integer.

    Raises:
        ValueError: if `s` is not a syntactically valid ISO-8601 duration,
            including (but not limited to) a missing leading "P", an empty
            "P"/"PT" body, a time component (H/M/S) appearing without the
            "T" separator, a fraction on anything but seconds, or any
            unrecognized character.
    """
    if not isinstance(s, str):
        raise ValueError(f"Duration must be a string, got {type(s)!r}")

    week_match = _WEEK_RE.fullmatch(s)
    if week_match:
        return int(week_match.group(1)) * 7 * 24 * 3600

    match = _DURATION_RE.fullmatch(s)
    if not match:
        raise ValueError(f"Invalid ISO-8601 duration: {s!r}")

    years = match.group("years")
    months = match.group("months")
    days = match.group("days")
    hours = match.group("hours")
    minutes = match.group("minutes")
    seconds = match.group("seconds")

    if "T" in s and hours is None and minutes is None and seconds is None:
        raise ValueError(f"Invalid ISO-8601 duration: {s!r}")

    if all(part is None for part in (years, months, days, hours, minutes, seconds)):
        raise ValueError(f"Invalid ISO-8601 duration: {s!r}")

    total = 0.0
    if years is not None:
        total += int(years) * 365 * 86400
    if months is not None:
        total += int(months) * 30 * 86400
    if days is not None:
        total += int(days) * 86400
    if hours is not None:
        total += int(hours) * 3600
    if minutes is not None:
        total += int(minutes) * 60
    if seconds is not None:
        total += float(seconds)

    return int(total) if total == int(total) else total
```

- Regex-driven validation naturally rejects fractions on anything but seconds, since only the seconds group's pattern (`\d+(?:\.\d+)?`) permits a decimal point.
- Time units (H/M/S) can only match inside the `T...` group, so any such unit appearing before "T" (or with no "T" at all) fails the full-string match and raises `ValueError`.
- "P" and "PT" are explicitly rejected by post-match checks: one catches a `T` with no H/M/S following it, the other catches the case where no component at all was captured.
- Weeks are parsed with a separate, stricter pattern (`P<digits>W` only) because the service guarantees weeks never combine with other units; any mixed form falls through and fails both patterns.
- The function always returns a plain `int` when the total is a whole number and a `float` only when fractional seconds make it necessary, per the `float | int` signature.
