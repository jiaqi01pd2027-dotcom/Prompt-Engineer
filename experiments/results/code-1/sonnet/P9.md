import re


def parse_iso_duration(s: str) -> float:
    """Convert an ISO-8601 duration string into a total number of seconds.

    Unit lengths are fixed (no calendar accuracy):
        1 year  = 365 days
        1 month = 30 days
        1 week  = 7 days
        1 day   = 24 hours

    Supported forms include "P1Y2M3DT4H5M6S", "P2W", "PT1.5S", "PT36H",
    and "P1DT12H". Weeks may only appear alone ("P2W" is valid, "P1W2D"
    is not). Fractional values are only permitted on the seconds
    component; a fraction on any other unit is invalid.

    Args:
        s: An ISO-8601 duration string.

    Returns:
        The total duration in seconds, as an int when the value is a
        whole number and a float otherwise.

    Raises:
        ValueError: if `s` is not a valid duration under the rules
            above -- e.g. a missing leading "P", an empty "P"/"PT", a
            time unit (H/M/S) given without the "T" separator, a
            fraction on a non-seconds unit, weeks combined with other
            units, or any unrecognized character.
    """
    if not isinstance(s, str) or not s.startswith("P"):
        raise ValueError(f"Invalid ISO-8601 duration (must start with 'P'): {s!r}")

    body = s[1:]
    if body == "":
        raise ValueError(f"Invalid ISO-8601 duration (empty 'P'): {s!r}")

    SECONDS_PER_DAY = 24 * 3600

    # Week-only form, e.g. "P2W". Weeks never combine with other units.
    if "W" in body:
        m = re.fullmatch(r"(\d+)W", body)
        if not m:
            raise ValueError(
                f"Invalid ISO-8601 duration (weeks must appear alone): {s!r}"
            )
        weeks = int(m.group(1))
        return weeks * 7 * SECONDS_PER_DAY

    has_t = "T" in body
    if has_t:
        parts = body.split("T")
        if len(parts) != 2:
            raise ValueError(
                f"Invalid ISO-8601 duration (multiple 'T' separators): {s!r}"
            )
        date_part, time_part = parts
        if time_part == "":
            raise ValueError(
                f"Invalid ISO-8601 duration ('T' with no time components): {s!r}"
            )
    else:
        date_part, time_part = body, ""

    date_match = re.fullmatch(r"(?:(\d+)Y)?(?:(\d+)M)?(?:(\d+)D)?", date_part)
    if date_match is None:
        raise ValueError(f"Invalid ISO-8601 duration (bad date part): {s!r}")

    time_match = re.fullmatch(
        r"(?:(\d+)H)?(?:(\d+)M)?(?:(\d+(?:\.\d+)?)S)?", time_part
    )
    if time_match is None:
        raise ValueError(f"Invalid ISO-8601 duration (bad time part): {s!r}")

    years, months, days = date_match.groups()
    hours, minutes, seconds = time_match.groups()

    if not any([years, months, days, hours, minutes, seconds]):
        raise ValueError(f"Invalid ISO-8601 duration (no components): {s!r}")

    total = 0.0
    if years:
        total += int(years) * 365 * SECONDS_PER_DAY
    if months:
        total += int(months) * 30 * SECONDS_PER_DAY
    if days:
        total += int(days) * SECONDS_PER_DAY
    if hours:
        total += int(hours) * 3600
    if minutes:
        total += int(minutes) * 60
    if seconds:
        total += float(seconds)

    if total == int(total):
        return int(total)
    return total
