import re

_WEEK_RE = re.compile(r'^P(\d+)W$')

_FULL_RE = re.compile(
    r'^P(?!$)'
    r'(?:(?P<years>\d+)Y)?'
    r'(?:(?P<months>\d+)M)?'
    r'(?:(?P<days>\d+)D)?'
    r'(?:T(?!$)'
    r'(?:(?P<hours>\d+)H)?'
    r'(?:(?P<minutes>\d+)M)?'
    r'(?:(?P<seconds>\d+(?:\.\d+)?)S)?'
    r')?$'
)


def parse_iso_duration(s):
    if not isinstance(s, str) or not s:
        raise ValueError(f"Invalid ISO-8601 duration: {s!r}")

    week_match = _WEEK_RE.match(s)
    if week_match:
        weeks = int(week_match.group(1))
        return weeks * 7 * 24 * 3600

    match = _FULL_RE.match(s)
    if not match:
        raise ValueError(f"Invalid ISO-8601 duration: {s!r}")

    years, months, days, hours, minutes, seconds_str = (
        match.group('years'), match.group('months'), match.group('days'),
        match.group('hours'), match.group('minutes'), match.group('seconds'),
    )

    if not any([years, months, days, hours, minutes, seconds_str]):
        raise ValueError(f"Invalid ISO-8601 duration: {s!r}")

    total = 0
    if years:
        total += int(years) * 365 * 24 * 3600
    if months:
        total += int(months) * 30 * 24 * 3600
    if days:
        total += int(days) * 24 * 3600
    if hours:
        total += int(hours) * 3600
    if minutes:
        total += int(minutes) * 60
    if seconds_str:
        total += float(seconds_str) if '.' in seconds_str else int(seconds_str)

    return total

How it satisfies each constraint:
- **Leading "P" required, "P"/"PT" alone rejected**: the `(?!$)` lookaheads right after `P` and right after `T` force at least one more character to follow; combined with an explicit "at least one component matched" check after the regex succeeds, bare `"P"` and `"PT"` (and `"P1YT"`, where `T` has nothing after it) all raise.
- **Time units require the T separator**: `H`/minute/`S` only appear inside the `(?:T...)?` group, so `"P1H"` or `"PT1D"` (D outside the T group) can't match and fall through to the `ValueError`.
- **Fractions only on seconds**: only the `seconds` group's pattern includes `(?:\.\d+)?`; every other unit is `\d+` only, so `"P1.5Y"` etc. fail to match.
- **Weeks handled separately** via `_WEEK_RE`, and deliberately mutually exclusive with other designators (`"P2W3D"` fails), matching ISO-8601 semantics.
- **No generic exceptions**: the only way out is a `ValueError` or a valid numeric return — no bare `Exception`, no silent `0`.
