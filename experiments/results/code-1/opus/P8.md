```python
from __future__ import annotations

import re
from decimal import Decimal

_ISO_DURATION_RE = re.compile(
    r"^P(?:(?P<years>\d+)Y)?(?:(?P<months>\d+)M)?(?:(?P<weeks>\d+)W)?"
    r"(?:(?P<days>\d+)D)?"
    r"(?:T(?:(?P<hours>\d+)H)?(?:(?P<minutes>\d+)M)?"
    r"(?:(?P<seconds>\d+(?:\.\d+)?)S)?)?$"
)

_UNIT_SECONDS = {
    "years": 365 * 86400,
    "months": 30 * 86400,
    "weeks": 7 * 86400,
    "days": 86400,
    "hours": 3600,
    "minutes": 60,
    "seconds": 1,
}


def parse_iso_duration(s: str) -> float | int:
    """Convert an ISO-8601 duration string into a total number of seconds.

    Unit assumptions (fixed by the billing service):
        1 year  = 365 days      1 day    = 24 hours
        1 month = 30 days       1 hour   = 60 minutes
        1 week  = 7 days        1 minute = 60 seconds

    Accepts P[nY][nM][nW][nD][T[nH][nM][nS]]. Weeks must stand alone, and only
    the seconds component may carry a decimal fraction. Returns an int when the
    total is whole, otherwise a float. Anything else -- a missing leading "P",
    an empty "P"/"PT", a time unit without the "T" separator, a fraction on any
    other unit, or an unknown character -- raises ValueError.
    """
    if not isinstance(s, str):
        raise ValueError(f"duration must be a string, got {type(s).__name__}")

    match = _ISO_DURATION_RE.match(s)
    if match is None:
        raise ValueError(f"invalid ISO-8601 duration: {s!r}")

    parts = {k: v for k, v in match.groupdict().items() if v is not None}
    if not parts:
        raise ValueError(f"duration has no components: {s!r}")
    if "T" in s and not ({"hours", "minutes", "seconds"} & parts.keys()):
        raise ValueError(f"time separator 'T' with no time components: {s!r}")
    if "weeks" in parts and len(parts) > 1:
        raise ValueError(f"weeks cannot be combined with other units: {s!r}")

    total = sum(
        (Decimal(value) * _UNIT_SECONDS[name] for name, value in parts.items()),
        Decimal(0),
    )
    return int(total) if total == total.to_integral_value() else float(total)
```

Assumptions

- Fixed calendar-free units: 1 year = 365 days, 1 month = 30 days, 1 week = 7 days, 1 day = 24 hours; no leap seconds or DST.
- Weeks are exclusive: `P2W` is valid, `P1W2D` raises, matching the "weeks always appear alone" rule.
- Only the seconds component may be fractional; the decimal is parsed with `Decimal` so `PT0.1S`-style values do not drift.
- Returns `int` for whole totals and `float` only when a fraction survives, so `PT36H` yields `129600`, not `129600.0`.
- Strict and uppercase-only: no sign, no lowercase designators, no surrounding whitespace, no `,` decimal separator; `"P"`, `"PT"`, `"P1S"` and `"P1.5D"` all raise `ValueError`.
