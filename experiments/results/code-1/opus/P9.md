```python
from __future__ import annotations

import re

# Two mutually exclusive shapes, so "P1W2D" cannot match structurally:
#   PnW                        (weeks always stand alone)
#   P[nY][nM][nD][T[nH][nM][nS]]
# Only the seconds field may carry a fraction. \Z (not $) so a trailing
# newline is not accepted.
_DURATION_RE = re.compile(
    r"""\AP(?:
            (?P<weeks>\d+)W
          |
            (?:(?P<years>\d+)Y)?
            (?:(?P<months>\d+)M)?
            (?:(?P<days>\d+)D)?
            (?:T
                (?:(?P<hours>\d+)H)?
                (?:(?P<minutes>\d+)M)?
                (?:(?P<seconds>\d+(?:\.\d+)?)S)?
            )?
        )\Z""",
    re.VERBOSE,
)

_SECONDS_PER = {
    "years": 365 * 24 * 3600,
    "months": 30 * 24 * 3600,
    "weeks": 7 * 24 * 3600,
    "days": 24 * 3600,
    "hours": 3600,
    "minutes": 60,
}


def parse_iso_duration(s: str) -> float | int:
    """Convert an ISO-8601 duration string into a total number of seconds.

    Accepted grammar:  P[nY][nM][nD][T[nH][nM][nS]]  or  PnW

    Fixed unit lengths: 1 year = 365 days, 1 month = 30 days,
    1 week = 7 days, 1 day = 24 hours. Weeks never combine with other units.
    A fraction is allowed only on the seconds field.

    Returns an int for a whole number of seconds and a float when the seconds
    field carries a fraction.

    Raises ValueError for anything malformed: a missing leading "P", an empty
    "P" or "PT", a time unit without the "T" separator ("P1S"), a fraction
    anywhere but on seconds, weeks mixed with other units, out-of-order or
    repeated units, and unknown characters.
    """
    if not isinstance(s, str):
        raise ValueError(f"duration must be a string, got {type(s).__name__}")

    match = _DURATION_RE.fullmatch(s)
    if match is None:
        raise ValueError(f"invalid ISO-8601 duration: {s!r}")

    parts = match.groupdict()

    # Every component is optional, so "P", "PT" and "P1DT" still match here.
    if all(value is None for value in parts.values()):
        raise ValueError(f"duration has no components: {s!r}")
    if "T" in s and all(parts[k] is None for k in ("hours", "minutes", "seconds")):
        raise ValueError(f"time separator 'T' with no time components: {s!r}")

    total = 0
    for name, factor in _SECONDS_PER.items():
        if parts[name] is not None:
            total += int(parts[name]) * factor

    seconds = parts["seconds"]
    if seconds is not None:
        # Stay exact for whole seconds; only widen to float for a real fraction.
        total += float(seconds) if "." in seconds else int(seconds)

    return total
```

Design notes:

- **Weeks as a separate alternation branch** rather than an optional `(\d+)W` group. `P1W2D` and `P2WT1H` then fail to match at all, instead of needing a follow-up "did weeks appear with anything else" check.
- **`P1S` / `P1H` rejection is structural** — H/M/S only exist inside the `T(...)` group, so a time unit without `T` never matches. Note that `M` is deliberately duplicated: in the date part it means months, in the time part minutes.
- **Two checks the regex can't express**, since every component is optional: an all-`None` match catches `P` and `PT`, and the `"T" in s` check catches `P1DT` and `P1YT`.
- **`\A`/`\Z` with `fullmatch`, not `^`/`$`.** I verified `$` accepts a trailing newline — `re.match(r'^P(\d+)D$', 'P1D\n')` matches — which would have let `"P1D\n"` through.
- **Return type stays `int` unless the seconds field has a `.`**, so integral durations avoid float rounding entirely; `PT1.5S` → `1.5`, `PT36H` → `129600` (int).

Two judgment calls worth confirming:

- **Comma decimal separator rejected.** ISO-8601 permits `PT1,5S`, but your spec listed only `.` and said unknown characters must raise. If you want it, change the seconds group to `\d+(?:[.,]\d+)?` and `.replace(",", ".")` before the float conversion.
- **Non-string input raises `ValueError`**, not `TypeError`, to match "invalid input raises ValueError". Drop the `isinstance` guard if you'd rather `None` and `5` raise `TypeError`.
