```python
import re
from decimal import Decimal

_ISO_DURATION_RE = re.compile(
    r"""^P(?!$)                       # leading P, not bare "P"
    (?:(?P<weeks>\d+)W$               # week form: weeks alone, nothing else
    |                                 # or the date/time form
      (?:(?P<years>\d+)Y)?
      (?:(?P<months>\d+)M)?
      (?:(?P<days>\d+)D)?
      (?:T(?!$)                       # T must be followed by at least one unit
        (?:(?P<hours>\d+)H)?
        (?:(?P<minutes>\d+)M)?
        (?:(?P<seconds>\d+(?:[.,]\d+)?)S)?
      )?
    )$""",
    re.VERBOSE,
)


def parse_iso_duration(s):
    """Convert an ISO-8601 duration string to a total number of seconds.

    Unit assumptions:
        1 year   = 365 days
        1 month  = 30 days
        1 week   = 7 days
        1 day    = 24 hours
        1 hour   = 60 minutes
        1 minute = 60 seconds

    Accepts full forms ("P1Y2M3DT4H5M6S"), the standalone week form ("P2W"),
    fractional seconds ("PT1.5S"), and any subset of components ("PT36H",
    "P1DT12H"). Returns an int when the total is whole, otherwise a float.

    Raises ValueError for anything that is not a valid ISO-8601 duration:
    a missing leading "P", a bare "P" or "PT" with no components, time units
    without the "T" separator, fractions on any unit other than seconds,
    weeks combined with other components, or unknown characters.
    """
    if not isinstance(s, str):
        raise ValueError(f"expected a string, got {type(s).__name__}")

    match = _ISO_DURATION_RE.match(s.strip())
    if match is None:
        raise ValueError(f"invalid ISO-8601 duration: {s!r}")

    parts = match.groupdict()
    if not any(parts.values()):
        raise ValueError(f"invalid ISO-8601 duration: {s!r}")

    seconds_per = {
        "weeks": 7 * 24 * 3600,
        "years": 365 * 24 * 3600,
        "months": 30 * 24 * 3600,
        "days": 24 * 3600,
        "hours": 3600,
        "minutes": 60,
        "seconds": 1,
    }

    total = Decimal(0)
    for unit, factor in seconds_per.items():
        value = parts[unit]
        if value is not None:
            total += Decimal(value.replace(",", ".")) * factor

    return int(total) if total == total.to_integral_value() else float(total)
```

Assumptions

- 1 year = 365 days and 1 month = 30 days exactly, so no calendar or leap-year arithmetic is performed; 1 week = 7 days, 1 day = 24 hours.
- The week designator is only valid alone (`P2W`); mixing `W` with other components raises `ValueError`.
- Fractions are permitted only on the seconds field, in either `.` or `,` decimal form; a fraction on any other unit is invalid.
- Only non-negative durations are accepted — a leading `-` or `+`, or per-component signs, raise `ValueError`.
- Returns `int` when the total has no fractional part and `float` otherwise; `Decimal` is used internally so values like `PT1.5S` are exact.
